常用函数导数与积分标准形式表

代数

\begin{align*} y &= x \\ \dfrac{dy}{dx} &= 1 \\ \int y\, dx &= \frac{1}{2} x^2 + C \\ \end{align*}
\begin{align*} y &= a \\ \dfrac{dy}{dx} &= 0 \\ \int y\, dx &= ax + C \\ \end{align*}
\begin{align*} y &= x ± a \\ \dfrac{dy}{dx} &= 1 \\ \int y\, dx &= \frac{1}{2} x^2 ± ax + C \\ \end{align*}
\begin{align*} y &= ax \\ \dfrac{dy}{dx} &= a \\ \int y\, dx &= \frac{1}{2} ax^2 + C \\ \end{align*}
\begin{align*} y &= x^2 \\ \dfrac{dy}{dx} &= 2x \\ \int y\, dx &= \frac{1}{3} x^3 + C \\ \end{align*}
\begin{align*} y &= x^n \\ \dfrac{dy}{dx} &= nx^{n-1} \\ \int y\, dx &= \dfrac{1}{n+1} x^{n+1} + C \\ \end{align*}
\begin{align*} y &= x^{-1} \\ \dfrac{dy}{dx} &= -x^{-2} \\ \int y\, dx &= \log_\epsilon x + C \\ \end{align*}
\begin{align*} y &= u ± v ± w \\ \dfrac{dy}{dx} &= \dfrac{du}{dx} ± \dfrac{dv}{dx} ± \dfrac{dw}{dx} \\ \int y\, dx &= \int u\, dx ± \int v\, dx ± \int w\, dx \\ \end{align*}
\begin{align*} y &= uv \\ \dfrac{dy}{dx} &= u\, \dfrac{dv}{dx} + v\, \dfrac{du}{dx} \\ \int y\, dx &= \text{No general form known} \\ \end{align*}
\begin{align*} y &= \dfrac{u}{v} \\ \dfrac{dy}{dx} &= \dfrac{v\, \dfrac{du}{dx} - u\, \dfrac{dv}{dx}}{v^2} \\ \int y\, dx &= \text{No general form known} \\ \end{align*}
\begin{align*} y &= u \\ \dfrac{dy}{dx} &= \dfrac{du}{dx} \\ \int y\, dx &= ux - \int x\, du + C \\ \end{align*}

指数与对数函数

\begin{align*} y &= \epsilon^x \\ \dfrac{dy}{dx} &= \epsilon^x \\ \int y\, dx &= \epsilon^x + C \\ \end{align*}
\begin{align*} y &= \log_\epsilon x \\ \dfrac{dy}{dx} &= x^{-1} \\ \int y\, dx &= x(\log_\epsilon x - 1) + C \\ \end{align*}
\begin{align*} y &= \log_{10} x \\ \dfrac{dy}{dx} &= 0.4343 × x^{-1} \\ \int y\, dx &= 0.4343x (\log_\epsilon x - 1) + C \\ \end{align*}
\begin{align*} y &= a^x \\ \dfrac{dy}{dx} &= a^x \log_\epsilon a \\ \int y\, dx &= \dfrac{a^x}{\log_\epsilon a} + C \\ \end{align*}

三角函数

\begin{align*} y &= \sin x \\ \dfrac{dy}{dx} &= \cos x \\ \int y\, dx &= -\cos x + C \\ \end{align*}
\begin{align*} y &= \cos x \\ \dfrac{dy}{dx} &= -\sin x \\ \int y\, dx &= \sin x + C \\ \end{align*}
\begin{align*} y &= \tan x \\ \dfrac{dy}{dx} &= \sec^2 x \\ \int y\, dx &= -\log_\epsilon \cos x + C \\ \end{align*}

圆形（反函数）

\begin{align*} y &= \arcsin x \\ \dfrac{dy}{dx} &= \dfrac{1}{\sqrt{(1-x^2)}} \\ \int y\, dx &= x · \arcsin x + \sqrt{1 - x^2} + C \\ \end{align*}
\begin{align*} y &= \arccos x \\ \dfrac{dy}{dx} &= -\dfrac{1}{\sqrt{(1-x^2)}} \\ \int y\, dx &= x · \arccos x - \sqrt{1 - x^2} + C \\ \end{align*}
\begin{align*} y &= \arctan x \\ \dfrac{dy}{dx} &= \dfrac{1}{1+x^2} \\ \int y\, dx &= x · \arctan x - \frac{1}{2} \log_\epsilon (1 + x^2) + C \\ \end{align*}

双曲函数

\begin{align*} y &= \sinh x \\ \dfrac{dy}{dx} &= \cosh x \\ \int y\, dx &= \cosh x + C \\ \end{align*}
\begin{align*} y &= \cosh x \\ \dfrac{dy}{dx} &= \sinh x \\ \int y\, dx &= \sinh x + C \\ \end{align*}
\begin{align*} y &= \tanh x \\ \dfrac{dy}{dx} &= \text{sech}^2 x \\ \int y\, dx &= \log_\epsilon \cosh x + C \\ \end{align*}

其他

\begin{align*} y &= \dfrac{1}{x + a} \\ \dfrac{dy}{dx} &= -\dfrac{1}{(x + a)^2} \\ \int y\, dx &= \log_\epsilon (x+a) + C \\ \end{align*}
\begin{align*} y &= \dfrac{1}{\sqrt{a^2 + x^2}} \\ \dfrac{dy}{dx} &= -\dfrac{x}{(a^2 + x^2)^{\frac{3}{2}}} \\ \int y\, dx &= \log_\epsilon (x + \sqrt{a^2 + x^2}) + C \\ \end{align*}
\begin{align*} y &= \dfrac{1}{a ± bx} \\ \dfrac{dy}{dx} &= \mp \dfrac{b}{(a ± bx)^2} \\ \int y\, dx &= ± \dfrac{1}{b} \log_\epsilon (a ± bx) + C \\ \end{align*}
\begin{align*} y &= \dfrac{a^2}{(a^2 + x^2)^{\frac{3}{2}}} \\ \dfrac{dy}{dx} &= -\dfrac{3a^2x}{(a^2 + x^2)^{\frac{5}{2}}} \\ \int y\, dx &= \dfrac{x}{\sqrt{a^2 + x^2}} + C \\ \end{align*}
\begin{align*} y &= \sin ax \\ \dfrac{dy}{dx} &= a · \cos ax \\ \int y\, dx &= -\dfrac{1}{a} \cos ax + C \\ \end{align*}
\begin{align*} y &= \cos ax \\ \dfrac{dy}{dx} &= -a · \sin ax \\ \int y\, dx &= \dfrac{1}{a} \sin ax + C \\ \end{align*}
\begin{align*} y &= \tan ax \\ \dfrac{dy}{dx} &= a · \sec^2ax \\ \int y\, dx &= -\dfrac{1}{a} \log_\epsilon \cos ax + C \\ \end{align*}
\begin{align*} y &= \sin^2 x \\ \dfrac{dy}{dx} &= \sin 2x \\ \int y\, dx &= \dfrac{x}{2} - \dfrac{\sin 2x}{4} + C \\ \end{align*}
\begin{align*} y &= \cos^2 x \\ \dfrac{dy}{dx} &= -\sin 2x \\ \int y\, dx &= \dfrac{x}{2} + \dfrac{\sin 2x}{4} + C \\ \end{align*}
\begin{align*} y &= \sin^n x \\ \dfrac{dy}{dx} &= n · \sin^{n-1} x · \cos x \\ \int y\, dx &= -\frac{\cos x}{n} \sin^{n-1} x + \frac{n-1}{n} \int \sin^{n-2} x\, dx + C \\ \end{align*}
\begin{align*} y &= \dfrac{1}{\sin x} \\ \dfrac{dy}{dx} &= -\dfrac{\cos x}{\sin^2 x} \\ \int y\, dx &= \log_\epsilon \tan \dfrac{x}{2} + C \\ \end{align*}
\begin{align*} y &= \dfrac{1}{\sin^2 x} \\ \dfrac{dy}{dx} &= -\dfrac{\sin 2x}{\sin^4 x} \\ \int y\, dx &= -\text{cotan} x + C \\ \end{align*}
\begin{align*} y &= \dfrac{1}{\sin x · \cos x} \\ \dfrac{dy}{dx} &= \dfrac{\sin^2 x - \cos^2 x}{\sin^2 x · \cos^2 x} \\ \int y\, dx &= \log_\epsilon \tan x + C \\ \end{align*}
\begin{align*} y &= \sin mx · \sin nx \\ \dfrac{dy}{dx} &= n · \sin mx · \cos nx + m · \sin nx · \cos mx \\ \int y\, dx &= \frac{1}{2} \cos(m - n)x - \frac{1}{2} \cos(m + n)x + C \\ \end{align*}
\begin{align*} y &= \sin^2 ax \\ \dfrac{dy}{dx} &= 2a·\sin 2ax \\ \int y\, dx &= \dfrac{x}{2} - \dfrac{\sin 2ax}{4a} + C \\ \end{align*}
\begin{align*} y &= \cos^2 ax \\ \dfrac{dy}{dx} &= -2a·\sin 2ax \\ \int y\, dx &= \dfrac{x}{2} + \dfrac{\sin 2ax}{4a} + C \\ \end{align*}
